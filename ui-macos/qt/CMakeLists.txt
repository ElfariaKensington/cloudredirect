cmake_minimum_required(VERSION 3.20)
project(CloudRedirectMacUI VERSION 2.6.6 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
find_package(Qt6 REQUIRED COMPONENTS Core Quick QuickControls2 Network)
set(UPSTREAM_UI ${CMAKE_CURRENT_SOURCE_DIR}/../../ui-linux)
qt_add_executable(CloudRedirect main.cpp backend_extra.cpp deployer.cpp ${UPSTREAM_UI}/src/backend.cpp ${UPSTREAM_UI}/src/backend.h ${UPSTREAM_UI}/src/deployer.h ${UPSTREAM_UI}/src/oauthservice.cpp ${UPSTREAM_UI}/src/oauthservice.h)
set_target_properties(CloudRedirect PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
target_include_directories(CloudRedirect PRIVATE . ${UPSTREAM_UI}/src ${CMAKE_CURRENT_SOURCE_DIR}/../../src/platform/macos)
target_compile_definitions(CloudRedirect PRIVATE CR_VERSION="2.6.6-macos")
target_link_libraries(CloudRedirect PRIVATE Qt6::Quick Qt6::QuickControls2 Qt6::Network)
set(qmlFiles Main.qml pages/SetupPage.qml pages/DashboardPage.qml pages/AppsPage.qml pages/BackupsPage.qml pages/CloudProviderPage.qml pages/StatsPage.qml pages/MigrationPage.qml)
foreach(file IN LISTS qmlFiles)
    set(path ${UPSTREAM_UI}/qml/${file})
    set_source_files_properties(${path} PROPERTIES QT_RESOURCE_ALIAS qml/${file})
    list(APPEND sources ${path})
endforeach()
qt_add_qml_module(CloudRedirect URI CloudRedirect VERSION 1.0 RESOURCE_PREFIX / QML_FILES ${sources})
enable_testing()
add_executable(macos_ui_backend_tests backend_tests.cpp backend_extra.cpp ${UPSTREAM_UI}/src/backend.cpp ${UPSTREAM_UI}/src/backend.h)
target_include_directories(macos_ui_backend_tests PRIVATE . ${UPSTREAM_UI}/src ${CMAKE_CURRENT_SOURCE_DIR}/../../src/platform/macos)
target_compile_definitions(macos_ui_backend_tests PRIVATE CR_VERSION="2.6.6-macos")
target_link_libraries(macos_ui_backend_tests PRIVATE Qt6::Core Qt6::Gui Qt6::Network)
add_test(NAME macos_ui_backend_tests COMMAND macos_ui_backend_tests)
set_tests_properties(macos_ui_backend_tests PROPERTIES TIMEOUT 30)

set_source_files_properties(qml/SteamCloudPage.qml PROPERTIES QT_RESOURCE_ALIAS qml/pages/SteamCloudPage.qml)
qt_add_resources(CloudRedirect macos_pages PREFIX /CloudRedirect FILES qml/SteamCloudPage.qml)

set_source_files_properties(qml/ManifestPage.qml PROPERTIES QT_RESOURCE_ALIAS qml/pages/ManifestPage.qml)
qt_add_resources(CloudRedirect manifest_page PREFIX /CloudRedirect FILES qml/ManifestPage.qml)
